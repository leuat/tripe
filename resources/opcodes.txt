# opcodes used in tripe


# address: place the variable at a specific memory address. no initial value
address8,0xF6,
address16,0xF7,

# uint types
uint8,0xF8,
uint16,0xF9,
uint32,0xFA,
uint64,0xFB,
int8,0xFC,
int16,0xFD,
int32,0xFE,
int64,0xFF,

# pointer types: *uint8, *uint16 or *uint32
ptr8,0xE0,
ptr16,0xE1,
ptr32,0xE2,

# section markers
.tripe,0x01,ival
.code,0x02,ival:p
.data,0x3,ival

# declare a variable
decl,0x4,p:ival
const,0x5,p:ival

# asm
.asm,0x6,p
.endasm,0x7,

# define cpu
.processor,0x8,p

#labels
.label,0x9,p

# where to place tripe generated code (mul/div code etc)
.gcode,0xA,0

# binary symbol table
.symbol,0xB,0

# block data stream

.uint8,0xF0,datastream
.uint16,0xF1,datastream
.int8,0xF2,datastream
.int16,0xF3,datastream
.incbin,0xF5,p

#end code
.end,0x0F,0

# the tree main opcodes in tripe

mov,0x12,p:p
store,0x13,p:p:p 
load,0x14,p:p:p 

return,0x10,0
rti,0x11,0

# binary operations

add,0x20,p:p:p
sub,0x21,p:p:p
mulu,0x28,p:p:p
divu,0x29,p:p:p
shl,0x2A,p:p:p
shr,0x2B,p:p:p
or,0x2C,p:p:p
and,0x2D,p:p:p
xor,0x2E,p:p:p

# branching

jeq,0x40,p:p:p
jneq,0x41,p:p:p
jgt,0x42,p:p:p
jlt,0x43,p:p:p
jgte,0x44,p:p:p
jlte,0x45,p:p:p
# signed versions
jgts,0x46,p:p:p
jlts,0x47,p:p:p
jgtes,0x48,p:p:p
jltes,0x49,p:p:p

# jumps

jump,0x80,p
call,0x81,p
